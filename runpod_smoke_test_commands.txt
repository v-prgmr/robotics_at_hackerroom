# RunPod ManiFlow demo train command buffer
#
# Goal of this demo:
# 1. Pull the latest Orbit repo.
# 2. Verify /workspace exists.
# 3. Install/repair the ManiFlow environment.
# 4. Force a fresh dataset download from Hugging Face Hub.
# 5. Run one tiny train/val pass.
# 6. Save one checkpoint: checkpoints/latest.ckpt.
# 7. Upload that checkpoint to a Hugging Face model repo.
# 8. Stop the RunPod pod after successful upload.
#
# Before pasting this into the RunPod terminal, replace every <...> value below.
# Copy/paste one section at a time if that is easier in the RunPod terminal.

# SECTION 1: pull latest code, inspect volume, and install/repair env.

cd /workspace/orbit
git pull

df -h /workspace
du -h -d 1 /workspace | sort -h

bash maniflow_orbit_bridge/runpod_setup_maniflow_env.sh

# SECTION 2: exports. Replace every <...> value before pasting this section.
# Do not commit real tokens into README or tracked files.

export TOKENIZERS_PARALLELISM="false"

# Hugging Face auth. Use a token with read access to the dataset repo and write access to the model repo.
export HF_TOKEN="<your-hf-token>"

# Dataset repo to download from. This must be a Hugging Face dataset repo.
export HF_DATASET_REPO_ID="<your-hf-dataset-repo>"
export HF_DATASET_REPO_TYPE="dataset"
export HF_DATASET_REVISION="main"
export HF_DATASET_PATH_IN_REPO="teabags_kitting_50_v2_maniflow.zarr"
export DATASET_NAME="teabags_kitting_50_v2_maniflow.zarr"
export HF_DATASET_FORCE_DOWNLOAD="true"

# Demo run settings. This is intentionally tiny but still writes a checkpoint.
export RUN_NAME="runpod_demo_train_once"
export OUTPUT_DIR="/workspace/outputs/train/${RUN_NAME}"
export DEBUG="False"
export BATCH_SIZE="4"
export NUM_WORKERS="1"
export NUM_EPOCHS="1"
export LOGGING_MODE="disabled"

# Artifact upload repo. This should be a Hugging Face model repo, not the dataset repo.
export PUSH_TO_HF="true"
export PUSH_TO_HF_ON_SUCCESS_ONLY="true"
export HF_REPO_ID="<your-hf-model-repo>"
export HF_REPO_TYPE="model"
export HF_PRIVATE="false"
export HF_REMOTE_PREFIX="runs/${RUN_NAME}"
export HF_UPLOAD_ALL_CHECKPOINTS="false"
export HF_UPLOAD_WANDB="false"

# Stop the RunPod pod after successful training and HF upload.
export STOP_POD_ON_EXIT="true"
export STOP_POD_ON_SUCCESS_ONLY="true"
export RUNPOD_API_KEY="<your-runpod-api-key>"
export RUNPOD_POD_ID="${POD_ID:-<your-pod-id>}"

# SECTION 3: safety check. This exits before training if any placeholder remains.

# Safety check: stop before doing anything expensive if placeholders were not replaced.
python - <<'PY'
import os
import sys

required = [
    "HF_TOKEN",
    "HF_DATASET_REPO_ID",
    "HF_REPO_ID",
    "RUNPOD_API_KEY",
    "RUNPOD_POD_ID",
]

bad = [name for name in required if not os.environ.get(name) or "<" in os.environ[name] or ">" in os.environ[name]]
if bad:
    print("Replace these placeholders before running:", ", ".join(bad))
    sys.exit(1)

print("Demo config OK")
print("Dataset repo:", os.environ["HF_DATASET_REPO_ID"])
print("Model repo:", os.environ["HF_REPO_ID"])
print("Output dir:", os.environ["OUTPUT_DIR"])
print("RunPod pod:", os.environ["RUNPOD_POD_ID"])
PY

# SECTION 4: clean only this demo output and run the demo.
# On success, this uploads latest.ckpt to HF Hub and stops the pod.

# Clean only this demo run output and force the dataset download test.
rm -rf "${OUTPUT_DIR}"
rm -rf "/workspace/dataset/${DATASET_NAME}"
mkdir -p /workspace/dataset /workspace/outputs/train

# This runs one short train/val pass, saves latest.ckpt, uploads latest.ckpt, then stops the pod.
bash maniflow_orbit_bridge/runpod_train_maniflow_orbit.sh \
  training.resume=false \
  training.num_epochs=1 \
  training.max_train_steps=1 \
  training.max_val_steps=1 \
  training.checkpoint_every=1 \
  training.val_every=1 \
  checkpoint.save_ckpt=true \
  checkpoint.save_last_ckpt=true \
  checkpoint.save_last_snapshot=false \
  checkpoint.topk.k=0
